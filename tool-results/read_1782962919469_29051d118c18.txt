   106→function ResultDisplay({
   107→  label,
   108→  value,
   109→  max,
   110→  suffix = "",
   111→  decimals = 2,
   112→  hint,
   113→}: {
   114→  label: string;
   115→  value: number;
   116→  max?: number;
   117→  suffix?: string;
   118→  decimals?: number;
   119→  hint?: string;
   120→}) {
   121→  const pct = max ? (value / max) * 100 : undefined;
   122→  const [celebrate, setCelebrate] = useState(false);
   123→  const prefersReduced = useReducedMotion();
   124→
   125→  return (
   126→    <GradientCard gradient="plum" className="p-6 sm:p-8 text-center">
   127→      <p className="text-xs uppercase tracking-widest opacity-80 font-semibold">
   128→        {label}
   129→      </p>
   130→      <div className="mt-3 flex items-baseline justify-center gap-2">
   131→        <span className="text-6xl sm:text-7xl font-bold tabular-nums tracking-tight">
   132→          <AnimatedCounter key={value} value={value} decimals={decimals} />
   133→        </span>
   134→        {suffix && <span className="text-2xl font-medium opacity-80">{suffix}</span>}
   135→      </div>
   136→      {pct !== undefined && (
   137→        <p className="text-sm opacity-80 mt-2">
   138→          {pct.toFixed(1)}% {max === 10 ? "efficiency" : "of max"}
   139→        </p>
   140→      )}
   141→      {hint && <p className="text-sm opacity-90 mt-2">{hint}</p>}
   142→      <AnimatePresence>
   143→        {celebrate && !prefersReduced && (
   144→          <motion.div
   145→            className="absolute inset-0 pointer-events-none"
   146→            initial={{ opacity: 1 }}
   147→            animate={{ opacity: 0 }}
   148→            transition={{ duration: 1.2, delay: 0.4 }}
   149→            onAnimationComplete={() => setCelebrate(false)}
   150→          >
   151→            {Array.from({ length: 12 }).map((_, i) => (
   152→              <motion.span
   153→                key={i}
   154→                className="absolute size-2 rounded-full"
   155→                style={{
   156→                  background: ["#fff", "#F59E0B", "#E11D48"][i % 3],
   157→                  left: "50%",
   158→                  top: "50%",
   159→                }}
   160→                initial={{ x: 0, y: 0, scale: 0 }}
   161→                animate={{
   162→                  x: Math.cos((i / 12) * 2 * Math.PI) * 120,
   163→                  y: Math.sin((i / 12) * 2 * Math.PI) * 120,
   164→                  scale: [0, 1, 0],
   165→                }}
   166→                transition={{ duration: 0.9 }}
   167→              />
   168→            ))}
   169→          </motion.div>
   170→        )}
   171→      </AnimatePresence>
   172→      {/* Expose setter via ref-like pattern */}
   173→      <CelebrationTrigger onCompute={setCelebrate} />
   174→    </GradientCard>
   175→  );
   176→}
   177→
   178→function CelebrationTrigger({ onCompute }: { onCompute: (v: boolean) => void }) {
   179→  // no-op; we trigger through the parent button via custom event
   180→  useEffect(() => {
   181→    const handler = () => onCompute(true);
   182→    window.addEventListener("ktu:celebrate", handler);
   183→    return () => window.removeEventListener("ktu:celebrate", handler);
   184→  }, [onCompute]);
   185→  return null;
   186→}
   187→
   188→import { useEffect } from "react";
   189→
   190→function fireCelebration() {
   191→  window.dispatchEvent(new Event("ktu:celebrate"));
   192→}

   479→function AttendanceCalculator() {

   489→      "ATTENDANCE",

   495→      kind: r.atRisk ? "warning" : "success",

   500→    });
   501→  };
   502→
   503→  return (
   504→    <div className="grid lg:grid-cols-3 gap-4">
   505→      <div className="lg:col-span-2 space-y-3">

   527→                min={1}

   765→function LabeledInput({
   766→  label,
   767→  sub,
   768→  value,
   769→  onChange,
   770→  max,
   771→}: {
   772→  label: string;
   773→  sub?: string;
   774→  value: number;
   775→  onChange: (v: number) => void;
   776→  max?: number;
   777→}) {
   778→  return (
   779→    <div>
   780→      <Label className="text-xs uppercase tracking-wider text-muted-foreground">
   781→        {label} {sub && <span className="text-muted-foreground/70 normal-case">({sub})</span>}
   782→      </Label>
   783→      <Input
   784→        type="number"
   785→        min={0}
   786→        max={max}
   787→        value={value}
   788→        onChange={(e) => onChange(Math.max(0, Number(e.target.value)))}
   789→        className="mt-1 h-12 text-lg bg-background"
   790→      />
   791→    </div>
   792→  );
   793→}

   804→function HistoryPanel({ type }: { type: string }) {

   806→  const entries = useMemo(
   807→    () => allEntries.filter((e) => e.type === type),
   808→    [allEntries, type],
   809→  );
   810→  if (entries.length === 0) {
   811→    return (
   812→      <GlassCard className="p-4 text-center">
   813→        <History className="size-5 mx-auto text-muted-foreground/60 mb-1" />
   814→        <p className="text-xs text-muted-foreground">No history yet</p>
   815→      </GlassCard>
   816→    );
   817→  }
   818→  return (
   819→    <GlassCard className="p-4">
   820→      <div className="flex items-center gap-2 mb-2">
   821→        <History className="size-4 text-muted-foreground" />
   822→        <h4 className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   823→          History
   824→        </h4>
   825→      </div>
   826→      <div className="space-y-1 max-h-48 overflow-y-auto">
   827→        {entries.slice(0, 8).map((e) => (

   837→    </GlassCard>
   838→  );
