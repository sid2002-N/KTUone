   106→   106→function ResultDisplay({

   109→   109→  max,

   112→   112→  hint,
   113→   113→}: {
   114→   114→  label: string;

   116→   116→  max?: number;
   117→   117→  suffix?: string;
   118→   118→  decimals?: number;
   119→   119→  hint?: string;
   120→   120→}) {
   121→   121→  const pct = max ? (value / max) * 100 : undefined;
   122→   122→  const [celebrate, setCelebrate] = useState(false);
   123→   123→  const prefersReduced = useReducedMotion();
   124→   124→
   125→   125→  return (
   126→   126→    <GradientCard gradient="plum" className="p-6 sm:p-8 text-center">

   128→   128→        {label}
   129→   129→      </p>
   130→   130→      <div className="mt-3 flex items-baseline justify-center gap-2">
   131→   131→        <span className="text-6xl sm:text-7xl font-bold tabular-nums tracking-tight">
   132→   132→          <AnimatedCounter key={value} value={value} decimals={decimals} />
   133→   133→        </span>
   134→   134→        {suffix && <span className="text-2xl font-medium opacity-80">{suffix}</span>}
   135→   135→      </div>
   136→   136→      {pct !== undefined && (
   137→   137→        <p className="text-sm opacity-80 mt-2">
   138→   138→          {pct.toFixed(1)}% {max === 10 ? "efficiency" : "of max"}
   139→   139→        </p>
   140→   140→      )}
   141→   141→      {hint && <p className="text-sm opacity-90 mt-2">{hint}</p>}
   142→   142→      <AnimatePresence>
   143→   143→        {celebrate && !prefersReduced && (
   144→   144→          <motion.div
   145→   145→            className="absolute inset-0 pointer-events-none"
   146→   146→            initial={{ opacity: 1 }}
   147→   147→            animate={{ opacity: 0 }}
   148→   148→            transition={{ duration: 1.2, delay: 0.4 }}
   149→   149→            onAnimationComplete={() => setCelebrate(false)}
   150→   150→          >
   151→   151→            {Array.from({ length: 12 }).map((_, i) => (
   152→   152→              <motion.span
   153→   153→                key={i}
   154→   154→                className="absolute size-2 rounded-full"
   155→   155→                style={{
   156→   156→                  background: ["#fff", "#F59E0B", "#E11D48"][i % 3],
   157→   157→                  left: "50%",
   158→   158→                  top: "50%",
   159→   159→                }}
   160→   160→                initial={{ x: 0, y: 0, scale: 0 }}
   161→   161→                animate={{
   162→   162→                  x: Math.cos((i / 12) * 2 * Math.PI) * 120,
   163→   163→                  y: Math.sin((i / 12) * 2 * Math.PI) * 120,
   164→   164→                  scale: [0, 1, 0],
   165→   165→                }}
   166→   166→                transition={{ duration: 0.9 }}
   167→   167→              />
   168→   168→            ))}
   169→   169→          </motion.div>
   170→   170→        )}
   171→   171→      </AnimatePresence>
   172→   172→      {/* Expose setter via ref-like pattern */}
   173→   173→      <CelebrationTrigger onCompute={setCelebrate} />
   174→   174→    </GradientCard>
   175→   175→  );
   176→   176→}
   177→   177→
   178→   178→function CelebrationTrigger({ onCompute }: { onCompute: (v: boolean) => void }) {
   179→   179→  // no-op; we trigger through the parent button via custom event
   180→   180→  useEffect(() => {
   181→   181→    const handler = () => onCompute(true);
   182→   182→    window.addEventListener("ktu:celebrate", handler);
   183→   183→    return () => window.removeEventListener("ktu:celebrate", handler);
   184→   184→  }, [onCompute]);
   185→   185→  return null;
   186→   186→}
   187→   187→
   188→   188→import { useEffect } from "react";
   189→   189→
   190→   190→function fireCelebration() {
   191→   191→  window.dispatchEvent(new Event("ktu:celebrate"));
   192→   192→}
